1//! 最小Steiner木
2//!
3//! # References
4//! - <https://www.slideshare.net/slideshow/ss-12131479/12131479>
5
6use std::collections::VecDeque;
7
8use crate::{algo::enum_bit::subset_asc::subset_asc, graph::*};
9
10#[derive(Clone, Copy)]
11enum Restore {
12    Merge(usize),
13    Edge(usize, usize),
14    None,
15}
16
17/// 最小Steiner木を求める。
18///
19/// - `graph` : $n$頂点の無向グラフ
20/// - `terminals` : `graph`の頂点集合の大きさ$k$の部分集合
21///
22/// **Time complexity** $O(3^k n + 2^k n^2)$
23pub fn min_steiner_tree<I>(
24    graph: &Graph<Undirected, u64, I>,
25    terminals: Vec<usize>,
26) -> Vec<&Edge<u64, I>> {
27    let n = graph.len();
28    let k = terminals.len();
29
30    let mut g: Vec<Vec<Option<&Edge<_, _>>>> = vec![vec![None; n]; n];
31    for (u, gu) in g.iter_mut().enumerate() {
32        for e in graph.node_of(u) {
33            let v = e.to();
34            if u != v && gu[v].is_none_or(|e2| e2.weight() > e.weight()) {
35                gu[v] = Some(e);
36            }
37        }
38    }
39
40    let mut dp: Vec<Vec<Option<u64>>> = vec![vec![None; n]; 1 << k];
41    let mut from: Vec<Vec<Restore>> = vec![vec![Restore::None; n]; 1 << k];
42
43    for u in 0..n {
44        dp[0][u] = Some(0);
45    }
46    for (i, &u) in terminals.iter().enumerate() {
47        for e in g[u].iter().flatten() {
48            dp[1 << i][e.to()] = Some(e.weight());
49            from[1 << i][e.to()] = Restore::Edge(e.to(), u);
50        }
51        dp[1 << i][u] = Some(0);
52    }
53
54    for s in 0_usize..1 << k {
55        for u in 0..n {
56            for t in subset_asc(s) {
57                if t == 0 || t == s {
58                    continue;
59                }
60                if let (Some(y), Some(z)) = (dp[t][u], dp[s ^ t][u])
61                    && dp[s][u].is_none_or(|x| x > y + z)
62                {
63                    dp[s][u] = Some(y + z);
64                    from[s][u] = Restore::Merge(t);
65                }
66            }
67        }
68
69        let mut q = VecDeque::new();
70        for u in 0..n {
71            if let Some(x) = dp[s][u] {
72                q.push_back((u, x));
73            }
74        }
75
76        while let Some((u, y)) = q.pop_front() {
77            if dp[s][u].is_some_and(|x| x < y) {
78                continue;
79            }
80
81            for e in g[u].iter().flatten() {
82                if dp[s][e.to()].is_none_or(|x| x > y + e.weight()) {
83                    dp[s][e.to()] = Some(y + e.weight());
84                    from[s][e.to()] = Restore::Edge(e.to(), u);
85                    q.push_back((e.to(), y + e.weight()));
86                }
87            }
88        }
89    }
90
91    let s = (1 << k) - 1;
92    let (u, _) = dp[s]
93        .iter()
94        .enumerate()
95        .filter_map(|(i, c)| c.map(|c| (i, c)))
96        .min_by_key(|(_, c)| *c)
97        .unwrap();
98
99    let mut ret = vec![];
100    let mut stack = vec![(s, u)];
101
102    while let Some((s, u)) = stack.pop() {
103        match from[s][u] {
104            Restore::Merge(t) => {
105                stack.push((t, u));
106                stack.push((s ^ t, u));
107            }
108            Restore::Edge(_u, v) => {
109                assert_eq!(u, _u);
110                stack.push((s, v));
111                ret.push(g[u][v].unwrap());
112            }
113            Restore::None => {}
114        }
115    }
116
117    ret
118}
